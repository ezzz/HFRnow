Bonjour,
 
Pour faire suite à mon problème de reboot aléatoire de mon PC, je me suis dis qu'une commande uptime serait quand même bien pratique , parce que "systeminfo" en ligne de commande c'est pas pratique, faut scroller en haut de la fenêtre à chaque fois pour avoir l'heure de démarrage du système...
 
J'ai donc fais le programme suivant :
 

Code :
  1. #include "windows.h"
  3. #include <iostream>
  4. #include <chrono>
  5. #include <format>
  7. int main()
  8. {
  9.     // 0. Définir la page de code de la console en UTF-8 (65001)
  10.     SetConsoleOutputCP(65001);
  11.     SetConsoleCP(65001);
  13.     // 1. Récupération du nombre de millisecondes écoulées depuis le démarrage du système
  14.     auto uptime = std::chrono::milliseconds(GetTickCount64());
  16.     // 2. Calcul de la date de démarrage du système
  17.     auto boot_time = std::chrono::system_clock::now() - uptime;
  18.     // 2.1 Suppression des microsecondes
  19.     auto boot_time_no_s = std::chrono::floor<std::chrono::seconds>(boot_time);
  21.     // 3. Affichage de la date et l'heure de démarrage du système
  22.     std::cout << "Heure de démarrage du système:\t" << std::format("{:%Y-%m-%d %H:%M:%S}", std::chrono::current_zone()->to_local(boot_time_no_s)) << std::endl;
  24.     // 4. Calcul de l'uptime en jour, heures, minutes, secondes
  25.     auto uptime_secondes = uptime.count() / 1000;
  26.     auto up_days  = uptime_secondes / 86400;
  27.     auto up_hours = uptime_secondes % 86400 / 3600;
  28.     auto up_min   = uptime_secondes % 86400 % 3600 / 60;
  29.     auto up_sec   = uptime_secondes % 86400 % 3600 % 60;
  31.     // 5. Affichage
  32.     switch (up_days)
  33.     {
  34.     case 0:
  35.         std::cout << "Uptime :\t\t\t" << up_hours << " heures, " << up_min << " minutes, " << up_sec << " secondes" << std::endl;
  36.         break;
  37.     case 1:
  38.         std::cout << "Uptime :\t\t\t" << up_days << " jour, " << up_hours << " heures, " << up_min << " minutes, " << up_sec << " secondes" << std::endl;
  39.         break;
  41.     default:
  42.         std::cout << "Uptime :\t\t\t" << up_days << " jours, " << up_hours << " heures, " << up_min << " minutes, " << up_sec << " secondes" << std::endl;
  43.         break;
  44.     }
  45. }

  1. #include "windows.h"
  3. #include <iostream>
  4. #include <chrono>
  5. #include <format>
  7. int main()

  11.     SetConsoleCP(65001);
  13.     // 1. Récupération du nombre de millisecondes écoulées depuis le démarrage du système
  14.     auto uptime_ms = std::chrono::milliseconds(GetTickCount64());
  16.    
  17.     // 2. Calcul de la date de démarrage du système
  18.     auto now = std::chrono::system_clock::now();
  19.     auto boot_time_ms    = now - uptime_ms;
  20.     // 2.1 Suppression des microsecondes
  21.     auto now_s           = std::chrono::floor<std::chrono::seconds>(now);
  22.     auto boot_time_s     = std::chrono::floor<std::chrono::seconds>(boot_time_ms);
  23.     // 2.2 Récupération de l'heure locale suivant le fuseau horaire
  24.     auto now_local       = std::chrono::current_zone()->to_local(now_s);
  25.     auto boot_time_local = std::chrono::current_zone()->to_local(boot_time_s);
  28.     // 3. Calcul de l'uptime en jour, heures, minutes, secondes
  29.     auto uptime_seconds = uptime_ms.count() / 1000;
  30.     auto up_days  = uptime_seconds / 86400;
  31.     auto up_hours = uptime_seconds % 86400 / 3600;
  32.     auto up_mins  = uptime_seconds % 86400 % 3600 / 60;
  33.     auto up_secs  = uptime_seconds % 86400 % 3600 % 60;
  35.    
  36.     // 4. Affichage
  37.     std::cout << "Heure de démarrage du système\t: " << std::format("{:%d/%m/%Y %H:%M:%S}", boot_time_local) << std::endl;
  38.     std::cout << "Heure actuelle\t\t\t: " << std::format("{:%d/%m/%Y %H:%M:%S}", now_local) << std::endl;
  39.     std::cout << "Démarré depuis\t\t\t: " << std::format("{:02}", up_days) << ":" << std::format("{:02}", up_hours) << ":" << std::format("{:02}", up_mins) << ":" << std::format("{:02}", up_secs);
  42.     //5. Sortie
  43.     return 0;
  44. }
